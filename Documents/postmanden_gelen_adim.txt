













Given(/^Mars kullanicisi ID listesini consola yazdirir$/, async function () {

    const axios = require('axios');

    async function getBlogs() {
        let config = {
            method: 'get',
            maxBodyLength: Infinity,
            url: 'https://qa.instulearn.com/api/blogs',
            headers: {
                'Accept': 'application/json',
                'x-api-key': '1234'
            }
        };

        try {
            const response = await axios.request(config);
            return response.data; // Testlerinizde doğrulamak (assert) için veriyi dönüyoruz
        } catch (error) {
            console.error("API İsteği Sırasında Hata Oluştu:", error);
            throw error; // Hatanın testi fail etmesi için yukarı fırlatıyoruz
        }
    }

    module.exports = { getBlogs };



});

    // 1. Bir önceki adımda 'this.response' içerisine kaydettiğimiz yanıtı alıyoruz

    if (responseData && Array.isArray(responseData)) {
